int main() {
    int i = 0;
    int count = 0;
    while (i < 3) {
        int j = 0;
        while (j < 3) {
            j = j + 1;
            if (j == 2) {
                continue;
            }
            count = count + 1;
        }
        i = i + 1;
    }
    printf("%d\n", count);
    return 0;
}
