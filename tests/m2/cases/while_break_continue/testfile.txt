int main() {
    int i = 0;
    int sum = 0;
    while (i < 10 && sum < 100) {
        i = i + 1;
        if (i == 5) {
            continue;
        }
        if (i == 9) {
            break;
        }
        sum = sum + i;
    }
    i = 0;
    while (i < 1) {
        i = i + 1;
    }
    printf("%d\n", sum);
    return 0;
}
